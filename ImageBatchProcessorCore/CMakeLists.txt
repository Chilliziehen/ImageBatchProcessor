set(SPV_FILES)
foreach(NAME process analyze)
 set(OUT ${CMAKE_BINARY_DIR}/shaders/${NAME}.spv)
 add_custom_command(OUTPUT ${OUT} COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/shaders COMMAND Vulkan::glslc --target-env=vulkan1.1 -O ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${NAME}.comp -o ${OUT} DEPENDS shaders/${NAME}.comp shaders/common.glsl VERBATIM)
 list(APPEND SPV_FILES ${OUT})
endforeach()
add_custom_target(ImageBatchProcessorShaders ALL DEPENDS ${SPV_FILES})
add_library(ImageBatchProcessorCore SHARED src/core.cpp src/io.cpp src/arguments.cpp)
add_dependencies(ImageBatchProcessorCore ImageBatchProcessorShaders)
target_include_directories(ImageBatchProcessorCore PUBLIC include)
target_compile_definitions(ImageBatchProcessorCore PRIVATE IBP_BUILD NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS IBP_SHADER_DIR="${CMAKE_BINARY_DIR}/shaders")
target_link_libraries(ImageBatchProcessorCore PRIVATE Vulkan::Vulkan raw windowscodecs ole32 propsys)
if(MSVC)
 target_compile_options(ImageBatchProcessorCore PRIVATE /W4 /EHsc /utf-8)
endif()
