cmake_minimum_required(VERSION 3.24)
project(ImageBatchProcessor VERSION 1.1.0 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "ImageBatchProcessor currently requires Windows (WIC file I/O).")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/LibRaw/libraw/libraw.h")
 message(FATAL_ERROR "LibRaw is missing. Run: git submodule update --init --recursive")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
find_package(Vulkan REQUIRED COMPONENTS glslc)
find_package(OpenMP)
file(GLOB_RECURSE RAW_SOURCES CONFIGURE_DEPENDS external/LibRaw/src/*.cpp)
list(FILTER RAW_SOURCES EXCLUDE REGEX "_ph\\.cpp$")
add_library(raw STATIC ${RAW_SOURCES})
target_include_directories(raw PUBLIC external/LibRaw)
target_compile_definitions(raw PUBLIC LIBRAW_NODLL PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
 target_compile_options(raw PRIVATE /MP /utf-8)
endif()
if(OpenMP_CXX_FOUND)
 target_link_libraries(raw PUBLIC OpenMP::OpenMP_CXX)
endif()
add_subdirectory(ImageBatchProcessorCore)
add_subdirectory(imgbp)
add_executable(imgbp-shaderc ShaderCompiler/main.cpp)
target_compile_definitions(imgbp-shaderc PRIVATE GLSLC_PATH=L"${Vulkan_GLSLC_EXECUTABLE}")
enable_testing()
add_test(NAME core_gpu_contract COMMAND imgbp selftest)
install(TARGETS ImageBatchProcessorCore imgbp imgbp-shaderc RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY ImageBatchProcessorCore/include/ DESTINATION include)
install(DIRECTORY ${CMAKE_BINARY_DIR}/shaders/ DESTINATION bin/shaders)
install(FILES README.md FORMAT.md QA_RESULTS.md CHANGELOG.md LICENSE THIRD_PARTY_NOTICES.md natural.imgarg DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES external/LibRaw/LICENSE.CDDL external/LibRaw/LICENSE.LGPL external/LibRaw/COPYRIGHT DESTINATION licenses/LibRaw)
set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
set(CMAKE_INSTALL_OPENMP_LIBRARIES TRUE)
include(InstallRequiredSystemLibraries)
option(IBP_BUILD_STUDIO "Build the SDL3/ImGui editor" ON)
if(IBP_BUILD_STUDIO)
 if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/imgui/imgui.h" OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/SDL/CMakeLists.txt")
  message(FATAL_ERROR "Studio dependencies missing. Run: git submodule update --init --recursive, or use -DIBP_BUILD_STUDIO=OFF")
 endif()
 add_subdirectory(Studio)
endif()
