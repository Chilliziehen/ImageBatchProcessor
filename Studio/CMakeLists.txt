set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(../external/SDL ${CMAKE_BINARY_DIR}/sdl EXCLUDE_FROM_ALL)
set(IMGUI_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../external/imgui)
add_library(imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_sdl3.cpp ${IMGUI_DIR}/backends/imgui_impl_vulkan.cpp)
target_include_directories(imgui PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends)
target_link_libraries(imgui PUBLIC SDL3::SDL3-static Vulkan::Vulkan)
add_executable(ImageBatchProcessorStudio main.cpp editor.cpp)
target_link_libraries(ImageBatchProcessorStudio PRIVATE imgui ImageBatchProcessorCore)
target_compile_definitions(ImageBatchProcessorStudio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
 target_compile_options(ImageBatchProcessorStudio PRIVATE /utf-8 /W4)
 target_link_options(ImageBatchProcessorStudio PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
endif()
install(TARGETS ImageBatchProcessorStudio RUNTIME DESTINATION bin)
install(FILES ../external/imgui/LICENSE.txt DESTINATION licenses/imgui)
install(FILES ../external/SDL/LICENSE.txt DESTINATION licenses/SDL)
